## 🎯 Принципы качества кода

### 1. Строгая типизация (Type Safety First)

**Принцип**: Все функции, переменные и возвращаемые значения должны иметь явные аннотации типов.

```python
# ✅ ПРАВИЛЬНО: Строгая типизация
from typing import Dict, List, Optional, Union, Any, Protocol

def process_blob_data(
    blob_obj: Any, 
    encoding: str = "utf-16"
) -> Dict[str, Union[str, int, None]]:
    """
    Обрабатывает BLOB данные с проверкой типов
    """
    if not isinstance(blob_obj, bytes):
        return {"content": str(blob_obj), "type": "string", "length": 0}
    
    try:
        content = blob_obj.decode(encoding)
        return {"content": content, "type": encoding, "length": len(content)}
    except UnicodeDecodeError:
        return {"content": blob_obj.hex(), "type": "binary", "length": len(blob_obj)}

# ❌ НЕПРАВИЛЬНО: Отсутствие типизации
def process_blob_data(blob_obj, encoding="utf-16"):
    return {"content": blob_obj.decode(encoding)}
```

### 2. Защитное программирование (Defensive Programming)

**Принцип**: Всегда проверять типы и наличие атрибутов перед их использованием.

```python
# ✅ ПРАВИЛЬНО: Защитное программирование
def safe_get_blob_content(blob_obj: Any) -> Optional[str]:
    """
    Безопасное извлечение содержимого BLOB
    """
    if not hasattr(blob_obj, "value"):
        return None
    
    value = blob_obj.value
    if not isinstance(value, bytes):
        return str(value) if value is not None else None
    
    # Пробуем разные кодировки
    for encoding in ["utf-16", "utf-8", "cp1251"]:
        try:
            return value.decode(encoding)
        except UnicodeDecodeError:
            continue
    
    return value.hex()

# ❌ НЕПРАВИЛЬНО: Прямое обращение к атрибутам
def get_blob_content(blob_obj):
    return blob_obj.value.decode("utf-16")
```

### 2. Error Handling Strategy

```python
# ✅ ПРАВИЛЬНО: Комплексная обработка ошибок
from functools import wraps
from typing import Callable, TypeVar, Optional

T = TypeVar('T')

def robust_error_handling(func: Callable[..., T]) -> Callable[..., Optional[T]]:
    """
    Декоратор для надежной обработки ошибок
    """
    @wraps(func)
    def wrapper(*args, **kwargs) -> Optional[T]:
        try:
            return func(*args, **kwargs)
        except StopIteration:
            # Нормальное завершение итератора
            return None
        except BrokenPipeError:
            # Нормальное завершение при использовании head
            return None
        except UnicodeDecodeError as e:
            # Ошибка кодировки - пробуем другую
            print(f"Encoding error: {e}")
            return None
        except Exception as e:
            # Другие ошибки
            print(f"Unexpected error: {e}")
            return None
    return wrapper
```

```python
class ProcessInput(BaseModel):

    metadata: Dict[str, str] = Field(description="Processing metadata")

# MCP Tool с автоматической валидацией
@mcp.tool()
async def process_data(input_data: ProcessInput) -> ProcessResult:
    """
    Process data with structured input and output
    """
    try:
        # Обработка данных

        return ProcessResult(
            status="success",
            result=result,
            metadata={"processed_at": datetime.now().isoformat()}

    except Exception as e:
        return ProcessResult(
            status="error",
            result="",
            metadata={"error": str(e)}
        )
```

```python
# ✅ ПРАВИЛЬНО: Асинхронное программирование

import aiohttp
from typing import List, Dict, Any

async def fetch_data_async(urls: List[str]) -> List[Dict[str, Any]]:
    """Асинхронное получение данных"""
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_single_url(session, url) for url in urls]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # Фильтруем исключения
        return [result for result in results if not isinstance(result, Exception)]

async def fetch_single_url(session: aiohttp.ClientSession, url: str) -> Dict[str, Any]:

            return await response.json()
    except Exception as e:
        return {"error": str(e), "url": url}
```

```python
from contextlib import asynccontextmanager
from typing import AsyncGenerator

@asynccontextmanager
async def database_connection() -> AsyncGenerator[DatabaseConnection, None]:
    """Асинхронный контекстный менеджер для БД"""
    connection = None
    try:
        connection = await create_database_connection()
        yield connection
    except Exception as e:

            await connection.rollback()
        raise e
```

### 6. Error Handling и Recovery

```python
# ✅ ПРАВИЛЬНО: Комплексная обработка ошибок
from typing import TypeVar, Callable, Optional
from functools import wraps
import asyncio

T = TypeVar('T')

def retry_on_failure(max_retries: int = 3, delay: float = 1.0):
    """Декоратор для повторных попыток"""
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        async def wrapper(*args, **kwargs) -> T:
            last_exception = None
            
            for attempt in range(max_retries):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < max_retries - 1:
                        await asyncio.sleep(delay * (2 ** attempt))  # Exponential backoff
                        continue
                    break
            
            raise last_exception
        return wrapper
    return decorator

@retry_on_failure(max_retries=3, delay=1.0)
async def fetch_external_data(url: str) -> Dict[str, Any]:
    """Получение внешних данных с повторными попытками"""
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as response:
            response.raise_for_status()
            return await response.json()

# ❌ НЕПРАВИЛЬНО: Отсутствие обработки ошибок
async def fetch_data(url):
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as response:
            return await response.json()  # Может упасть
```

## 🎯 Специализированные требования для 1С

### 1. BLOB Processing Patterns

```python
# ✅ ПРАВИЛЬНО: Специализированная обработка BLOB для 1С
def enhanced_blob_extraction(blob_obj: Any) -> Dict[str, Any]:
    """
    Улучшенное извлечение BLOB с множественными методами для 1С
    """
    methods_used = []
    content = None
    
    # Метод 1: onec_dtools UTF-16 (стандарт 1С)
    if hasattr(blob_obj, "value") and isinstance(blob_obj.value, bytes):
        try:
            content = blob_obj.value.decode("utf-16")
            methods_used.append("onec_dtools_utf16")
        except UnicodeDecodeError:
            pass
    
    # Метод 2: onec_dtools UTF-8 (fallback)
    if not content:
        try:
            content = blob_obj.value.decode("utf-8")
            methods_used.append("onec_dtools_utf8")
        except UnicodeDecodeError:
            pass
    
    # Метод 3: CP1251 (русская кодировка)
    if not content:
        try:
            content = blob_obj.value.decode("cp1251")
            methods_used.append("cp1251")
        except UnicodeDecodeError:
            pass
    
    # Метод 4: Fallback hex
    if not content:
        content = blob_obj.value.hex()
        methods_used.append("fallback_hex")
    
    return {
        "content": content,
        "methods_used": methods_used,
        "success": len(methods_used) > 0,
        "encoding": methods_used[0] if methods_used else "unknown"
    }
```

### 2. 1C Data Extraction Patterns

```python
# ✅ ПРАВИЛЬНО: Правильное извлечение данных из 1С
def extract_1c_row_data(row: Any, include_blobs: bool = True) -> Dict[str, Any]:
    """
    Извлечение данных строки 1С с правильной обработкой BLOB
    """
    if include_blobs:
        # ВАЖНО: row.as_list(True) для включения BLOB полей
        row_data = row.as_list(True)
    else:
        row_data = row.as_dict()
    
    processed_data = {}
    
    for field_name, field_value in row_data.items():
        if isinstance(field_value, bytes):
            # BLOB поле - специальная обработка
            processed_data[field_name] = enhanced_blob_extraction(field_value)
        else:
            # Обычное поле
            processed_data[field_name] = {
                "value": str(field_value) if field_value is not None else None,
                "type": type(field_value).__name__
            }
    
    return processed_data
```

### 3. 1C Error Handling

```python
# ✅ ПРАВИЛЬНО: Специализированная обработка ошибок 1С
def handle_1c_errors(func: Callable[..., T]) -> Callable[..., Optional[T]]:
    """
    Декоратор для обработки специфичных ошибок 1С
    """
    @wraps(func)
    def wrapper(*args, **kwargs) -> Optional[T]:
        try:
            return func(*args, **kwargs)
        except ValueError as e:
            if "Unknown field type" in str(e):
                print(f"1C field type error: {e}")
                return None
            raise
        except BrokenPipeError:
            # Нормальное завершение при использовании head
            return None
        except StopIteration:
            # Нормальное завершение итератора
            return None
        except Exception as e:
            print(f"Unexpected 1C error: {e}")
            return None
    return wrapper
```

### 1. Type Guards

```python
# ✅ ПРАВИЛЬНО: Type Guards для безопасности
from typing import TypeGuard

def is_blob_field(value: Any) -> TypeGuard[bytes]:
    """Type guard для проверки BLOB полей"""
    return isinstance(value, bytes)

def is_string_field(value: Any) -> TypeGuard[str]:
    """Type guard для проверки строковых полей"""
    return isinstance(value, str)

def process_field(field_name: str, value: Any) -> Dict[str, Any]:
    """Безопасная обработка полей с type guards"""
    if is_blob_field(value):
        return enhanced_blob_extraction(value)
    elif is_string_field(value):
        return {"content": value, "type": "string", "length": len(value)}
    else:
        return {"content": str(value), "type": type(value).__name__, "length": 0}
```

### 2. Validation Patterns

```python
# ✅ ПРАВИЛЬНО: Валидация данных
from typing import List, Dict, Any, Optional

def validate_extraction_result(result: Any) -> bool:
    """Валидация результата извлечения"""
    if not isinstance(result, dict):
        return False
    
    required_keys = ["content", "type", "length"]
    return all(key in result for key in required_keys)

def safe_extract_blob(blob_obj: Any) -> Optional[Dict[str, Any]]:
    """Безопасное извлечение BLOB с валидацией"""
    result = enhanced_blob_extraction(blob_obj)
    
    if not validate_extraction_result(result):
        return None
    
    return result
```

### 3. Error Recovery Patterns

```python
# ✅ ПРАВИЛЬНО: Паттерны восстановления после ошибок
def resilient_blob_processing(blob_obj: Any) -> Dict[str, Any]:
    """Устойчивая обработка BLOB с множественными стратегиями"""
    strategies = [
        lambda x: x.decode("utf-16"),
        lambda x: x.decode("utf-8"),
        lambda x: x.decode("cp1251"),
        lambda x: x.hex()
    ]
    
    for i, strategy in enumerate(strategies):
        try:
            content = strategy(blob_obj)
            return {
                "content": content,
                "method": f"strategy_{i+1}",
                "success": True
            }
        except (UnicodeDecodeError, AttributeError):
            continue
    
    return {
        "content": "failed_to_decode",
        "method": "none",
        "success": False
    }
```
