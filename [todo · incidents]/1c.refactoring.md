```python
class BaseExtractor:

    def process_row(self, row, row_index: int) -> Dict
    def validate_data(self, data: Dict) -> bool
```

```python
class DocumentExtractor(BaseExtractor):

    def extract_table_parts(self, db, table_name: str, row_index: int) -> Dict
```
